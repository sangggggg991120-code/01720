-- UI BUILDER HELPER (BỘ CÔNG CỤ TẠO GIAO DIỆN)

local UI = {}

function UI.Notification(title, text, duration)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = title,
            Text = text,
            Duration = duration or 4
        })
    end)
end

function UI.AddCorner(parent, radius)
    local corner = Instance.new("UICorner", parent)
    corner.CornerRadius = UDim.new(0, radius or 8)
    return corner
end

function UI.AddStroke(parent, color, thickness)
    local stroke = Instance.new("UIStroke", parent)
    stroke.Color = color or Color3.fromRGB(60, 60, 70)
    stroke.Thickness = thickness or 1
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    return stroke
end

function UI.CreateText(parent, pos, sz, text, sizeFont, color, font)
    local lbl = Instance.new("TextLabel", parent)
    lbl.BackgroundTransparency = 1
    lbl.Position = pos
    lbl.Size = sz
    lbl.Text = text
    lbl.Font = font or Enum.Font.GothamBold
    lbl.TextSize = sizeFont or 12
    lbl.TextColor3 = color or Color3.fromRGB(240, 240, 240)
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    return lbl
end

function UI.CreateBtn(parent, pos, sz, text, bg, colorText, sizeFont)
    local btn = Instance.new("TextButton", parent)
    btn.Position = pos
    btn.Size = sz
    btn.Text = text
    btn.BackgroundColor3 = bg or Color3.fromRGB(35, 35, 45)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = sizeFont or 12
    btn.TextColor3 = colorText or Color3.fromRGB(255, 255, 255)
    btn.AutoButtonColor = true
    UI.AddCorner(btn, 6)
    return btn
end

function UI.CreateScrollingFrame(parent, sz, vis)
    local sf = Instance.new("ScrollingFrame", parent)
    sf.BackgroundTransparency = 1
    sf.Size = sz
    sf.Visible = vis
    sf.ScrollBarThickness = 3
    sf.ScrollBarImageColor3 = Color3.fromRGB(0, 170, 255)
    sf.ScrollingDirection = Enum.ScrollingDirection.Y
    sf.CanvasSize = UDim2.new(0, 0, 0, 0)
    return sf
end

function UI.SetupAutoScroll(sf, layout, padding)
    layout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        sf.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + (padding or 15))
    end)
end

local function applyLayout(parent)
    local layout = Instance.new("UIListLayout", parent)
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Padding = UDim.new(0, 8)
    return layout
end

local function isEnemyNPC(obj)
    if not obj or not obj.Parent then return false end
    if Players:GetPlayerFromCharacter(obj) then return false end
    
    local hum = obj:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    
    local name = string.lower(obj.Name)
    local parentName = string.lower(obj.Parent.Name)
    
    if string.find(name, "base") or string.find(name, "bank") or string.find(name, "crystal") or string.find(name, "tumore") or string.find(name, "tower") or string.find(name, "spawn") then 
        return false 
    end
    if string.find(parentName, "unit") or string.find(parentName, "player") or string.find(parentName, "tower") then 
        return false 
    end
    
    return true
end

local attPlayer, attMonster, tracerBeam = Instance.new("Attachment"), Instance.new("Attachment"), Instance.new("Beam")
tracerBeam.Width0, tracerBeam.Width1 = 0.2, 0.2
tracerBeam.Color = ColorSequence.new(Color3.fromRGB(255, 0, 0))

tracerBeam.Attachment0, tracerBeam.Attachment1 = attPlayer, attMonster

local function resetMonsters()
    if tracerBeam then tracerBeam.Parent = nil end
    for monster, data in pairs(originalData) do
        if monster and monster.Parent and data.Root then
            pcall(function()
                data.Root.Size = data.Size
                data.Root.Transparency = data.Transparency
                data.Root.CanCollide = data.CanCollide
                data.Root.CFrame = data.CFrame
                monster:PivotTo(data.CFrame)
            end)
        end
    end
    table.clear(originalData)
    table.clear(CurrentValidMonsters)
end

task.spawn(function()
    while task.wait(0.1) do
        if AutoplayEnabled then
            pcall(function()
                local myChar = LP.Character
                local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
                if not myRoot or IsResetting then
                    if tracerBeam then tracerBeam.Parent = nil end
                    return
                end
                
                local currentTime = os.clock()
                local validMonsters = {}
                
                for _, obj in pairs(workspace:GetDescendants()) do
                    if obj:IsA("Model") and obj ~= myChar and isEnemyNPC(obj) then
                        local root = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Head")
                        if root then
                            if not monsterSpawnTimes[obj] then monsterSpawnTimes[obj] = currentTime end
                            if (currentTime - monsterSpawnTimes[obj]) >= 0.5 then
                                if not originalData[obj] then
                                    originalData[obj] = {
                                        Root = root,
                                        CFrame = root.CFrame,
                                        Size = root.Size,
                                        Transparency = root.Transparency,
                                        CanCollide = root.CanCollide
                                    }
                                end
                                if (originalData[obj].CFrame.Position - myRoot.Position).Magnitude <= DetectRange then
                                    table.insert(validMonsters, obj)
                                end
                            end
                        end
                    end
                end
                CurrentValidMonsters = validMonsters
                
                if #validMonsters > 0 and ActiveMode ~= 2 then 
                    local effDist = (ForwardDistance == 0) and 1.5 or ForwardDistance
                    local targetCFrame = myRoot.CFrame * CFrame.new(0, -2, -effDist)
                    
                    for _, monster in ipairs(validMonsters) do
                        local root = monster:FindFirstChild("HumanoidRootPart") or monster:FindFirstChild("Head")
                        if root then
                            root.Size = HitboxSize
                            root.Transparency = 1
                            root.CanCollide = false
                            if root:IsA("BasePart") then root.AssemblyLinearVelocity = Vector3.new(0,0,0) end
                            monster:PivotTo(targetCFrame)
                        end
                    end
                    local firstMonsterRoot = validMonsters[1]:FindFirstChild("HumanoidRootPart") or validMonsters[1]:FindFirstChild("Head")
                    if firstMonsterRoot then
                        attPlayer.Parent = myRoot
                        attPlayer.Position = Vector3.new(0, 0, 0)
                        attMonster.Parent = firstMonsterRoot
                        attMonster.Position = Vector3.new(0, 0, 0)
                        if tracerBeam then tracerBeam.Parent = myRoot end
                    end
                elseif ActiveMode == 2 then
                    for _, monster in ipairs(validMonsters) do
                        local data = originalData[monster]
                        local root = monster:FindFirstChild("HumanoidRootPart") or monster:FindFirstChild("Head")
                        if data and root then
                            root.Size = data.Size
                            root.Transparency = data.Transparency
                            root.CanCollide = data.CanCollide
                        end
                    end
                    if tracerBeam then tracerBeam.Parent = nil end
                else
                    if tracerBeam then tracerBeam.Parent = nil end
                end
                
                for monster, _ in pairs(monsterSpawnTimes) do
                    if not monster:IsDescendantOf(workspace) then
                        monsterSpawnTimes[monster] = nil
                        originalData[monster] = nil
                    end
                end
            end)
        else
            if #CurrentValidMonsters > 0 or next(originalData) ~= nil then resetMonsters() end
        end
    end
end)

    AutoplayBtn.Text = AutoplayEnabled and "Autoplay: ON" or "Autoplay: OFF"
    
    if AutoplayEnabled then
        UI.Notification("hamsterscript", "Hãy đảm bảo Unit của bạn đã được trang bị!", 3)
        task.spawn(function()
            while AutoplayEnabled do
                local char = LP.Character
                if char and char:FindFirstChild("Humanoid") and char.Humanoid.Health > 0 and not IsResetting then
                    local myHrp = char:FindFirstChild("HumanoidRootPart")
                    if ActiveMode == 1 then
                        if (TARGET_POS - myHrp.Position).Magnitude >= 3 then
                            local finalTarget = FlyModeEnabled and Vector3.new(TARGET_POS.X, FlyYLevel, TARGET_POS.Z) or TARGET_POS
                            local t = TweenService:Create(myHrp, TweenInfo.new((finalTarget - myHrp.Position).Magnitude / 200, Enum.EasingStyle.Linear), {CFrame = CFrame.new(finalTarget)})
                            t:Play()
                            while t.PlaybackState == Enum.PlaybackState.Playing and AutoplayEnabled and not IsResetting do task.wait() end
                        end
                    elseif ActiveMode == 2 then
                        if #CurrentValidMonsters > 0 then
                            local targetMonster = CurrentValidMonsters[1]
                            local mRoot = targetMonster:FindFirstChild("HumanoidRootPart") or targetMonster:FindFirstChild("Head")
                            if mRoot and myHrp then
                                if string.find(string.lower(targetMonster.Name), "boss") then
                                    local bossTopPos = Vector3.new(mRoot.Position.X, FlyModeEnabled and FlyYLevel or (mRoot.Position.Y + 6), mRoot.Position.Z)
                                    myHrp.CFrame = CFrame.lookAt(bossTopPos, Vector3.new(mRoot.Position.X, bossTopPos.Y, mRoot.Position.Z))
                                else
                                    if os.clock() - LastTriangleTick >= 0.5 then TriangleIndex = TriangleIndex % 3 + 1 LastTriangleTick = os.clock() end
                                    local rad = math.rad((TriangleIndex - 1) * 120)
                                    local finalRange = (ForwardDistance > 0) and (3 + ForwardDistance) or 3
                                    local pointPos = mRoot.Position + Vector3.new(math.cos(rad) * finalRange, 0, math.sin(rad) * finalRange)
                                    local finalPointPos = FlyModeEnabled and Vector3.new(pointPos.X, FlyYLevel, pointPos.Z) or pointPos
                                    myHrp.CFrame = CFrame.lookAt(finalPointPos, Vector3.new(mRoot.Position.X, finalPointPos.Y, mRoot.Position.Z)) 
                                end
                            end
                        else
                            if myHrp then myHrp.CFrame = CFrame.new(FlyModeEnabled and Vector3.new(TARGET_POS.X, FlyYLevel, TARGET_POS.Z) or TARGET_POS) end
                        end
                    end
                
                    if AutoAttackEnabled and char:FindFirstChild("RemoteEvent") then
                        if #CurrentValidMonsters > 0 then
                            for _, monster in ipairs(CurrentValidMonsters) do
                                local r = monster:FindFirstChild("HumanoidRootPart") or monster:FindFirstChild("Head")
                                if r then char.RemoteEvent:FireServer(Enum.UserInputType.MouseButton1, r.Position, r.Position) end
                            end
                        else
                            char.RemoteEvent:FireServer(Enum.UserInputType.MouseButton1, Vector3.new(-22, 3, -134), Vector3.new(-3, 10, 62))
                        end
                    end
                    task.wait(AttackSpeed)
                else
                    LP.CharacterAdded:Wait()
                    task.wait(1.5)
                end
            end
        end)
    else
        resetMonsters()
    end
end)

-- ==========================================
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    
    if method ~= "FireServer" then
        return oldNamecall(self, ...)
    end
    
    local args = table.pack(...)
    local isMatchFound = false
    local detectedUnit = nil
    
    for i = 1, args.n do
        local arg = args[i]
        if type(arg) == "string" and string.find(string.lower(arg), "changechar") then
            isMatchFound = true
        elseif typeof(arg) == "Instance" then
            detectedUnit = arg
        end
    end
    
    if isMatchFound then
        CapturedArgs = args      
        CapturedRemote = self    
        CapturedUnitName = detectedUnit and tostring(detectedUnit.Name) or "Unknown Unit"
        
        -- Thêm Unit vừa bắt vào danh sách History List
        local newEntry = {
            Name = CapturedUnitName,
            Args = CapturedArgs,
            Remote = CapturedRemote
        }
        
        -- Lọc tránh trùng lặp
        for i, v in ipairs(CapturedUnitsList) do
            if v.Name == CapturedUnitName then
                table.remove(CapturedUnitsList, i)
                break
            end
        end
        table.insert(CapturedUnitsList, 1, newEntry) -- Chèn vào vị trí đầu tiên (Mới nhất)
        
        task.spawn(function()
            pcall(RefreshUnitListUI) -- Làm mới UI danh sách Unit
            pcall(function()
                if ResetSpawnBtn then
                    ResetSpawnBtn.Text = "[" .. CapturedUnitName .. "]: Gọi Spawn 1 Lần"
                    ResetSpawnBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 255)
                    ResetSpawnBtn.TextColor3 = Color3.new(1, 1, 1)
                end
            end)
        end)
    end
    
    return oldNamecall(self, ...)
end)
